FROM node:24-alpine

WORKDIR /app

COPY package*.json ./

RUN npm install --omit=dev

COPY . .

EXPOSE 3000

CMD ["node", "app.js"]

docker build -t my-node-app:v1 .
docker images
docker run -d --name my-node-app -p 3000:3000 my-node-app:v1
docker ps
docker inspect my-node-app
docker stop my-node-app
docker rm my-node-app
