SELECT
    status,
    COUNT(*) AS total_todos
FROM TODOS
GROUP BY status;

SELECT
    u.name,
    u.email,
    COUNT(t.id) AS todos_count
FROM users u
LEFT JOIN todos t ON u.id = t.user_id
GROUP BY u.id, u.name, u.email;

SELECT
    u.name,
    u.email
FROM users u
LEFT JOIN todos t ON u.id = t.user_id
WHERE t.id IS NULL;
